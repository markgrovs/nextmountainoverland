{{- /* Reusable gallery renderer — globs all images in the current page bundle. */ -}} {{- $imgs := .Resources.ByType "image" -}} {{- if $imgs -}}
{{- range sort $imgs "Name" "asc" -}} {{- $caption := "" -}} {{- $sidecar := printf "%s.md" (strings.TrimSuffix (path.Ext .Name) .Name) -}} {{- with $.Page.Resources.Get $sidecar -}} {{- $caption = .Content | plainify -}} {{- end -}} {{- $thumb := .Resize "600x q80" -}} {{- $full := .Resize "2048x q85" -}} {{- end -}}
{{- end -}}
