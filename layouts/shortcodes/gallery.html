{{- /* Gallery shortcode — page-bundle aware, PhotoSwipe lightbox. Usage (auto-collect all images in the current page bundle): {{< gallery >}} Usage (explicit glob pattern within the bundle): {{< gallery pattern="images/*.jpg" >}} Usage (filter to a subfolder, e.g. only the "wide" shots): {{< gallery pattern="images/wide/*" >}} Each image may have an optional sidecar .md file whose body becomes the PhotoSwipe caption. EXIF is preserved by Hugo's image pipeline. */ -}} {{- $pattern := .Get "pattern" | default "*" -}} {{- $imgs := .Page.Resources.Match $pattern -}} {{- $imgs = sort $imgs "Name" "asc" -}} {{- if not $imgs -}} {{- errorf "gallery shortcode in %q: no page resources matched pattern %q" .File.Path $pattern -}} {{- end -}}
{{- range $imgs -}} {{- if (not (in (slice ".md" ".json" ".yaml" ".yml" ".txt") (path.Ext .Name))) -}} {{- $caption := "" -}} {{- $sidecar := printf "%s.md" (strings.TrimSuffix (path.Ext .Name) .Name) -}} {{- with $.Page.Resources.Get $sidecar -}} {{- $caption = .Content | plainify -}} {{- end -}} {{- $thumb := .Resize "600x q80" -}} {{- $full := .Resize "2048x q85" -}} {{- end -}} {{- end -}}
